import { Unstyled } from '@storybook/blocks'
import { fontFamily, fontSize, fontWeight, typography } from '../../lib/ui/utils/typography.js'

<Unstyled>

<div class="flex flex-col gap-8">
  {Object.entries(typography).map(([className, { size, family, weight }]) => (
    <div key={className}>
      <span class="font-mono t-body-3-medium mb-3 text-waterloo">
        {fontFamily[family][0]} - {size} ({fontSize[size].join(' / ')}) - {weight} (
        {fontWeight[weight]}) - <code>.{className}</code>
      </span>
      <p class={'ml-5 ' + className}>The quick brown fox jumps over the lazy dog.</p>
    </div>
  ))}
  {Object.entries(fontSize).map(([text, [size, lineHeight]]) => (
    <div key={text}>
      <span class="font-mono t-body-3-medium mb-3 text-waterloo">
        {text} ({size} / {lineHeight}) - <code>.text-{text}</code>
      </span>
      <p class={'ml-5 ' + 'text-' + text}>The quick brown fox jumps over the lazy dog.</p>
    </div>
  ))}
</div>

</Unstyled>
