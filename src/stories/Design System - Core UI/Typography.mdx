import { Unstyled } from '@storybook/blocks'
import tailwindConfig from '../../../tailwind.config.js'
import { typography } from '../../lib/ui/utils/typography.js'

<Unstyled>

  {Object.entries(typography).map(([className, { size, family, weight }]) => (

        {tailwindConfig.theme.fontFamily[family][0]} -{' '}
        {tailwindConfig.theme.fontSize[size].join(' / ')} - {weight} - <code>.{className}</code>
      </span>
      <p class={'ml-5 ' + className}>The quick brown fox jumps over the lazy dog.</p>
    </div>
  ))}
  {Object.entries(tailwindConfig.theme.fontSize).map(([text, [fontSize, lineHeight]]) => (

        {fontSize} / {lineHeight} - <code>.text-{text}</code>
      </span>
      <p class={'ml-5 ' + 'text-' + text}>The quick brown fox jumps over the lazy dog.</p>
    </div>
  ))}
</div>

</Unstyled>
